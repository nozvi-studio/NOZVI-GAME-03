# P3b-3 Menu screens Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Port the D2 menu mockups (Boot, Home, Loadout, Settings, Results) to Phaser, add the new Settings scene (reachable from Home and as an overlay from the Pause menu), and generate all skill/number copy from `config/balance.json`. P4 meta elements (coins, marbles, skins, marble machine) are not rendered (spec D8).

**Architecture:** Every screen has a pure view model in `src/game/flow/` (unit-tested: texts, flags, states). Pure geometry (`src/game/ui/menuLayout.ts`: logo letters, snake-path smoothing, per-screen `layoutColumn` specs) is unit-tested for "fits at 540x960". Thin Phaser scenes read the view model and draw with the P3-0 widget kit; a few menu-only widgets live in `src/game/ui/menuWidgets.ts` and `icons.ts`. Settings are written immediately through the session save (`session.updateSettings`) and pushed to `setDisplayPrefs`.

**Tech Stack:** Phaser 3.90, TypeScript 7 strict, Vite 8, Vitest 5.

**Spec:** `docs/superpowers/specs/2026-11-07-p3-juice-screens-design.md` (§3 D1-D3, D5, D7, D8; §4 P3b-3). Mockups: `docs/design/mockups/html/{BootD2,HomeD2,LoadoutD2,SettingsD2,ResultsD2}.html` (primary), `docs/design/mockups/src/{HomeD2,ResultsD2}.dc.html` (props `firstRun`, `state`), `docs/design/mockups/png/*` (cross-check only). Game rules: GDD §3b/§10.3, MEC-007, MEC-008.

/** P3b3-R3 Home: top = settings gear row + logo; middle = hero, stepper, Play, chips, hint; bottom = hill peek. */
export function homeColumn(f: { compact: boolean; stepper: boolean; extras: boolean; hint: boolean }, minShrink: number): Spec {

export interface LogoLetter { ch: string; color: string; rotDeg: number; dy: number }
/** P3b3-R13 letter styles from the mockup: colour cycles from `colorOffset`, tilt and lift alternate; a space is a gap. */
export function logoLetters(text: string, colorOffset: number): LogoLetter[] {
  const cols = TOKENS.menu.logoColors;
  let n = 0;
  return [...text].map((ch, i) => {
    if (ch === ' ') return { ch, color: '', rotDeg: 0, dy: 0 };
    const idx = n++;
    return { ch, color: cols[(idx + colorOffset) % cols.length]!, rotDeg: i % 2 ? 3 : -3, dy: i % 2 ? 2 : -2 };
  });
}

export interface Pt { x: number; y: number }
/** P3b3-R13 HomeD2 snake trail: polyline with quadratic-rounded corners, sampled every ~step px. */
export function smoothPath(pts: readonly Pt[], r: number, step: number): Pt[] {
  const out: Pt[] = [pts[0]!];
  const line = (to: Pt): void => {
    const from = out[out.length - 1]!;
    const n = Math.max(1, Math.ceil(Math.hypot(to.x - from.x, to.y - from.y) / step));
    for (let i = 1; i <= n; i++) out.push({ x: from.x + (to.x - from.x) * i / n, y: from.y + (to.y - from.y) * i / n });
  };
  for (let i = 1; i < pts.length - 1; i++) {
    const p = pts[i - 1]!, c = pts[i]!, nx = pts[i + 1]!;
    const l1 = Math.hypot(p.x - c.x, p.y - c.y), l2 = Math.hypot(nx.x - c.x, nx.y - c.y);
    const rr = Math.min(r, l1 / 2, l2 / 2);
    const a = { x: c.x + (p.x - c.x) / l1 * rr, y: c.y + (p.y - c.y) / l1 * rr };
    const b = { x: c.x + (nx.x - c.x) / l2 * rr, y: c.y + (nx.y - c.y) / l2 * rr };
    line(a);
    const steps = Math.max(2, Math.ceil(rr * 1.6 / step));
    for (let s = 1; s <= steps; s++) {
      const u = s / steps;
      out.push({ x: (1 - u) ** 2 * a.x + 2 * (1 - u) * u * c.x + u * u * b.x, y: (1 - u) ** 2 * a.y + 2 * (1 - u) * u * c.y + u * u * b.y });
    }
  }
  line(pts[pts.length - 1]!);
  return out;
}

export type IconId = typeof ICON_IDS[number];
type G = Pick<Phaser.GameObjects.Graphics, 'fillStyle' | 'lineStyle' | 'fillCircle' | 'strokeCircle' | 'fillRoundedRect' | 'fillTriangle' | 'lineBetween' | 'beginPath' | 'moveTo' | 'lineTo' | 'strokePath' | 'closePath' | 'fillPath'>;

/** Draws `id` centred on (0,0) inside a `size` box. */
export function drawIcon(g: G, id: IconId, size: number, color: number): void {
  const u = size / 24, w = 2.4 * u;
  g.fillStyle(color); g.lineStyle(w, color);
  const poly = (pts: number[][], fill: boolean): void => {
    g.beginPath(); g.moveTo(pts[0]![0]! * u, pts[0]![1]! * u);
    for (const p of pts.slice(1)) g.lineTo(p[0]! * u, p[1]! * u);
    g.closePath(); if (fill) g.fillPath(); else g.strokePath();
  };
  switch (id) {
    case 'back': g.beginPath(); g.moveTo(3 * u, -8 * u); g.lineTo(-5 * u, 0); g.lineTo(3 * u, 8 * u); g.strokePath(); break;

import type { Balance, SkillId } from '../../core';
import type { LoadoutEditState } from './loadoutEdit';
export interface SkillCardView { id: SkillId; nameKey: string; locked: boolean; cost: number; unlockLevel: number;
  desc: { key: string; vars: Record<string, number> } | null; slot: 0 | 1 | null }
export interface LoadoutView { titleKey: 'loadout.title' | 'loadout.titleOne';
  slots: { index: 0 | 1; skill: SkillId | null; active: boolean }[]; cards: SkillCardView[]; startAt: number; energy: number | null }
const sec = (ms: number): number => ms / 1000;
// P3b3-R8 card copy is generated from balance (spec §2): no typed numbers.
export function skillDesc(id: SkillId, b: Balance): { key: string; vars: Record<string, number> } {
  const k = `skill.${id}.desc`;
  switch (id) {
    case 'slowmo': return { key: k, vars: { s: sec(b.skills.slowmo.durationMs) } };
    case 'tailSnip': return { key: k, vars: { s: sec(b.skills.tailSnip.wallLifeMs) } };
    case 'shield': return { key: k, vars: { s: sec(b.skills.shield.durationMs) } };
    case 'mangoRain': return { key: k, vars: { min: b.skills.mangoRain.dropsMin, max: b.skills.mangoRain.dropsMax, s: sec(b.skills.mangoRain.durationMs) } };
  }
}
// P3b3-R9 MEC-007-R3/R6 view: slot count follows the model (C3), locked cards show "Unlocks at Lx" (text) and no description.
export function loadoutView(e: LoadoutEditState, b: Balance, ctx: { level: number; carried: number }): LoadoutView {
  return {
    titleKey: e.slotCount === 2 ? 'loadout.title' : 'loadout.titleOne',
    slots: ([0, 1] as const).filter(i => i < e.slotCount).map(i => ({ index: i, skill: e.selection[i], active: e.activeSlot === i })),
    cards: b.loadout.order.map(id => {
      const locked = !e.unlocked.includes(id), s = e.selection.indexOf(id);
      return { id, nameKey: `skill.${id}.name`, locked, cost: b.skills[id].cost, unlockLevel: b.skills[id].unlockLevel,
        desc: locked ? null : skillDesc(id, b), slot: s === 0 || s === 1 ? s : null };
    }),
    startAt: ctx.level, energy: ctx.carried > 0 ? ctx.carried : null,
  };
}
