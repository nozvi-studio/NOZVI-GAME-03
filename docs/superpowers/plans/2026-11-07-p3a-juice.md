# P3a Juice Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement every GDD §10.4 feedback row except "Marble machine" (P4): eat mango, eat energy orb, skill cast, level up, biome change, checkpoint unlocked and death (shake + red flash + X eyes + "RUN OVER" card with a death line per cause, then Results), with new synth SFX, haptics behind the setting, reduced-motion variants and a particle/perf guard.

**Architecture:** A pure mapper `src/game/juice/juiceMap.ts` turns a sim/flow event into a `JuiceCue` (`board` visuals, `screen` visuals, `sounds`, `haptic`) using numbers from a new `balance.juice` block (validated in `parseBalance`). Reduced motion and the sound/haptics settings are applied **inside** the mapper, so every rule is unit-tested without Phaser. A thin `JuicePlayer` (HUD scene) calls the mapper once per event, plays sounds/haptics through the existing `Synth`, forwards `board` visuals to `GameScene.playBoardFx` (extended `Effects`) and `screen` visuals to a `ScreenFx` sink. HUD positions (energy bar, level bar, score, skill buttons, flag) are reached only through a small injected `JuiceHud` interface so P3b-2's HUD split does not matter.

**Tech Stack:** Phaser 3.90, TypeScript 7 strict, Vite 8, Vitest 5.

**Spec:** `docs/superpowers/specs/2026-11-07-p3-juice-screens-design.md` (§3 D6, D9; §4 P3a). GDD §10.2, §10.4, §3 accessibility. Mockups: `docs/design/mockups/html/DeathD2.html`, `BannersD2.html`, `SkillActiveD2.html` (variants in `../src/*.dc.html`).

```ts
// types.ts
import type { Balance, CauseKey, SimEvent, SkillId } from '../../core';
import type { SfxName } from '../audio/synth';
export type ColorId = 'food' | 'gold' | 'rock' | 'snake' | 'danger' | 'orb' | 'slowmo' | 'tailSnip' | 'shield' | 'mangoRain';
export interface Pt { x: number; y: number }
export type BoardVisual =
  | { kind: 'burst'; x: number; y: number; color: ColorId; count: number; speed: number; lifeMs: number }
  | { kind: 'floatText'; x: number; y: number; points: number; big: boolean }
  | { kind: 'gulp' } | { kind: 'snakePop'; ms: number }
  | { kind: 'levelGlow'; alpha: number; ms: number }
  | { kind: 'shake'; amplitudePx: number; ms: number }
  | { kind: 'tint'; color: ColorId; alpha: number; fadeMs: number; ms: number }
  | { kind: 'tintEnd'; fadeMs: number }
  | { kind: 'biomeSweep'; biome: number; alpha: number; ms: number; sweep: boolean };
export type HudTarget = 'score' | 'levelBar' | 'energyBar';
export type ScreenVisual =
  | { kind: 'sparks'; from: { x: number; y: number }; count: number; ms: number; staggerMs: number; color: ColorId }
  | { kind: 'barBump'; target: HudTarget; scale: number; ms: number }
  | { kind: 'iconPop'; slot: 0 | 1; color: ColorId; scale: number; ms: number }
  | { kind: 'flash'; color: ColorId; alpha: number; ms: number }
  | { kind: 'flagPlant'; level: number; ms: number }
  | { kind: 'deathFlash'; alpha: number; ms: number; vignette: boolean };
export interface JuiceCue {
  board: BoardVisual[]; screen: ScreenVisual[]; sounds: SfxName[]; haptic: number | number[] | null;
  death: { cause: CauseKey } | null;
}
export interface JuiceCtx { balance: Balance; sound: boolean; haptics: boolean; reduced: boolean }
export type JuiceInput = SimEvent | { type: 'checkpointUnlocked'; level: number };
/** HUD positions and bumps, injected by the HUD scene so juice never knows the HUD layout (P3b-2 safe). */
export interface JuiceHud {
  point(target: HudTarget): Pt;
  skillPoint(slot: 0 | 1): Pt;
  bump(target: HudTarget, scale: number, ms: number): void;
  plantFlag(level: number, ms: number): void;
}
```

```ts
import type { Balance, SkillId } from '../../core';
import type { SfxName } from '../audio/synth';
import type { BoardVisual, ColorId, JuiceCtx, JuiceCue, JuiceInput, ScreenVisual } from './types';

const SKILL_SFX: Record<SkillId, SfxName> = { slowmo: 'skillSlowmo', tailSnip: 'skillSnip', shield: 'skillShield', mangoRain: 'skillRain' };
const SKILL_COLOR: Record<SkillId, ColorId> = { slowmo: 'slowmo', tailSnip: 'tailSnip', shield: 'shield', mangoRain: 'mangoRain' };
const TINTED: ReadonlySet<SkillId> = new Set<SkillId>(['slowmo', 'shield']);

export interface DeathTimeline { shakeMs: number; flashMs: number; cardInMs: number; totalMs: number; skipLockMs: number }
/** P3a-R7 death sequence timing; total <= 1.5 s is enforced by parseBalance. */
export function deathTimeline(b: Balance, reduced: boolean): DeathTimeline {
  const d = b.juice.death, r = b.juice.reduced;
  return { shakeMs: reduced ? 0 : d.shakeMs, flashMs: reduced ? r.softMs : d.flashMs, cardInMs: d.cardInMs,
    totalMs: reduced ? r.deathTotalMs : d.totalMs, skipLockMs: d.skipLockMs };
}
/** P3a-R8 banner entrance: slide + scale normally, a soft fade when reduced. */
export function bannerMotion(b: Balance, reduced: boolean): { inMs: number; slidePx: number; scaleFrom: number } {

  return reduced ? { inMs: b.juice.reduced.softMs, slidePx: 0, scaleFrom: 1 } : { inMs: n.inMs, slidePx: n.slidePx, scaleFrom: n.scaleFrom };

      screen({ kind: 'deathFlash', alpha: r ? cap : j.death.flashAlpha, ms: r ? soft : j.death.flashMs, vignette: !r });
      cue.sounds = ['die']; cue.haptic = j.haptics.deathMs; cue.death = { cause: ev.cause }; break;
    case 'rotted': burst(ev.x, ev.y, 'rock', 'rot'); cue.sounds = ['rot']; break;
    case 'dust': burst(ev.x, ev.y, 'rock', 'dust'); break;
    case 'landed': cue.sounds = ['land']; break;
    case 'rampEnded': cue.sounds = ['go']; break;
    case 'shieldBroke': cue.sounds = ['shieldBreak']; break;
    case 'skillUnlocked': cue.sounds = ['unlock']; break;
    case 'runWon': cue.sounds = ['biome']; break;
    default: break;
  }
  if (!ctx.sound) cue.sounds = [];     // P3a-R9
  if (!ctx.haptics) cue.haptic = null; // P3a-R9
  return cue;
}
```

```ts
  handle(ev: JuiceInput): void {
    const s = this.d.settings();
    const cue = mapJuice(ev, { balance: this.d.balance, sound: s.sound, haptics: s.haptics, reduced: this.d.reduced() });

  runOver(won: boolean): void {
    if (this.dying || this.done) return;
    void won;
    this.timers.push(this.d.later(deathTimeline(this.d.balance, this.d.reduced()).totalMs, () => this.finish()));

      this.timers.push(this.d.later(tl.cardInMs + tl.skipLockMs, () => this.card?.armSkip(() => this.finish())));

    this.timers.push(this.d.later(tl.totalMs, () => this.finish()));
  }

    if (this.done) return;
    this.done = true; this.destroy(); this.d.onDone();

  destroy(): void { for (const t of this.timers) t.remove(); this.timers = []; this.card?.destroy(); this.card = null; }
```
