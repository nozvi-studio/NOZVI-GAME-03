## Findings (root cause of the quit gap; conflicts with mockups)
**Quit root cause (verified in code, 2026-11-07).** The pause overlay Quit path is already wired: `HudScene` Quit button -> `session.quit()` -> `endRun('quit')` (MEC-003-R20, since 2026-11-04) and `tests/session/session.test.ts` proves the log record at session level. The existing HUD test `MEC-003-R20 pause Quit ...` only checks `summary().cause`, so it never proved the stored log, the carry or the gate exclusion. The phone log "1 run quit without run_end" therefore came from a run that was left without pressing Quit (tab closed, reload, app killed, back gesture): `Session` holds the run in memory and writes `run_end` only from `endRun`; `hidden()`/`pause()` only flush the log and the provisional carry (MEC-008-R7). Two real gaps follow: (1) `startRun` while a previous run is still active silently orphans it (no `run_end`, no R6 carry); (2) nothing in the log reader tells an abandoned run from an in-progress one. Fix: Task 1 (TDD) adds a stored-data characterisation test for the pause Quit path, makes `startRun` end an active run with cause `quit`, and adds a read-only `abandonedRuns(log)` helper. Closing the tab cannot write a record; that case stays "provisional carry only" (MEC-008-R7, unchanged).

**Mockup vs spec/code conflicts** (spec wins; none changes a gameplay rule):
1. D-pad: current code shows the D-pad always (greybox, DECISIONS 2026-11-03); GDD §3 and MEC-001 say "optional D-pad (setting)", GameD2 has none, GameDpadD2 has it. Plan follows GDD + mockups: hidden unless `settings.dpad`.
2. Level-up chip "Your snake shrinks to 4 segments" is always shown in the mockup; MEC-001-R4 only drops cells when the snake is longer than the start length. Plan shows the chip only when the same event batch has `shrunk {from > to}`; the number comes from `to` (= `balance.snake.start.length`).
3. CountdownD2 makes the whole screen a link that starts the level (tap to skip). MEC-001-R2 / `runFlow` use a fixed `flow.countdownMs` and the first direction input starts movement. Not adopted.
4. BannersD2 example data is inconsistent with the specs (checkpoint "Next run starts at L6!" while HUD says Level 6; mud tooltip at Level 6 while mud starts at L7, MEC-003-R15). Treated as illustration; real values come from events and balance.
5. Skill-unlocked tooltip: mockup "New skill: Slow-mo" with a tail pointing at the button; MEC-007-R12 text also says which side to tap (`tip.skillUnlocked.slot`). The tail replaces the side word; the slot-less case (Shield at L11, both slots filled) keeps the existing "Pick it before your next run" text.
6. Mockup HUD has no Best score, no sound button, no "Energy x/100" text. Display-only: Best stays on Results; mute moves to Settings (P3b-3), Home keeps its sound toggle, the `M` key still works.
7. Pause mockup drops "You get a short countdown when you resume" and shows "You keep marbles ..." (P4). Resume still runs the countdown (`flowResume`, unchanged); marbles line hidden.
8. The mockups show no "Warming up N" ramp hint (MEC-008-R8 feedback). Kept: it replaces the checkpoint chip while the ramp is active.

```ts
// P3b2-R3 runs that logged a start but never an end (tab closed, reload, app killed). Read-only; the log schema is unchanged.
export function abandonedRuns(log: RunLog): number[] {
  const dbg = debugRunIds(log), ended = new Set<number>(), started: number[] = [];
  for (const r of log.entries) {
    if (dbg.has(r.runId)) continue;
    if (r.type === 'run_end') ended.add(r.runId);
    else if (r.type === 'run_start' && !started.includes(r.runId)) started.push(r.runId);
  }
  return started.filter(id => !ended.has(id)).sort((a, c) => a - c);
}
```

```ts
export interface StackItem { h: number; mt?: number }
export function stack(items: readonly StackItem[], o: { gap: number; padTop: number; padBottom: number }): { ys: number[]; height: number } {
```

```ts
import type { Balance, SimSnapshot, SkillId } from '../../core';
import { biomeName, levelTitle } from '../hudText';

export type T = (k: string, v?: Record<string, string | number>) => string;
export type HudSnap = Pick<SimSnapshot, 'level' | 'biome' | 'challenge' | 'progress' | 'need' | 'score' | 'energy' | 'energyMax' | 'skills' | 'rampActive' | 'rampMangoesLeft'>;
export type SkillState = 'off' | 'ready' | 'active';
export interface SkillView { slot: 0 | 1; id: SkillId; cost: number; state: SkillState; ring: number | null }
export interface HudView {
  levelTitle: string; biomeName: string; quota: { done: string; total: string };
  score: string; energy: { text: string; fill: number };
  skills: readonly [SkillView | null, SkillView | null];
  chip: { kind: 'ramp' | 'checkpoint'; text: string } | null;
}
const clamp01 = (n: number): number => Math.min(1, Math.max(0, n));
/** Total ms of a timed skill (ring denominator); Tail Snip is instant (its wall is shown by the pill), so no ring. */
const timedMs = (id: SkillId, b: Balance): number | null =>
  id === 'slowmo' ? b.skills.slowmo.durationMs : id === 'shield' ? b.skills.shield.durationMs : id === 'mangoRain' ? b.skills.mangoRain.durationMs : null;

/** P3b2-R7 MEC-003 §8 / MEC-006 §8 / MEC-007 §8 / MEC-008-R8, R11: everything the HUD shows, as plain data. */
export function hudView(s: HudSnap, hint: { key: string; level: number } | null, b: Balance, t: T): HudView {
  const skill = (slot: 0 | 1): SkillView | null => {
    const k = s.skills[slot];
    if (!k || k.id === null) return null;
    const total = timedMs(k.id, b);
    return { slot, id: k.id, cost: k.cost, state: k.active ? 'active' : k.affordable ? 'ready' : 'off',
      ring: k.active && total !== null ? clamp01(k.leftMs / total) : null };
  };
  const max = s.level >= b.level.maxLevel;
  return {
    levelTitle: levelTitle(s, t), biomeName: biomeName(s.biome, t),
    quota: max ? { done: t('hud.quotaMax'), total: '' } : { done: String(s.progress), total: `/${s.need}` },
    score: String(s.score), energy: { text: String(s.energy), fill: clamp01(s.energy / s.energyMax) },
    skills: [skill(0), skill(1)],
    chip: s.rampActive ? { kind: 'ramp', text: t('hud.ramp', { n: s.rampMangoesLeft }) }
      : hint ? { kind: 'checkpoint', text: t(hint.key, { level: hint.level }) } : null,
  };
}
```

```ts
  private debugKey: { c: Phaser.GameObjects.Container } | null = null; private debugNext: { c: Phaser.GameObjects.Container } | null = null;

  constructor() { super('Hud'); }

  create(): void {
    this.gameScene = this.scene.get('Game') as GameScene;
    this.session = this.registry.get('session') as Session; this.balance = this.registry.get('balance') as Balance;
    const save = this.session.save(), layout = hudLayout(true); // pad geometry always computed; visibility follows settings
    this.cpCount = save.unlockedCheckpoints.length; this.bestLevelAtStart = save.bestLevel;
    this.orbTipShown = false; this.leaving = false; this.heading = ''; this.shrunkTo = null;
    this.synth = createSynth(); this.synth.setOn(save.settings.sound); this.synth.unlock();
    this.input.setGlobalTopOnly(true);
    this.drawHill();
    this.top = new HudTop(this, layout, () => this.gameScene.apply(flowTogglePause(this.gameScene.flow, this.balance)));
    this.skills = new SkillButtons(this, layout, slot => this.useSkill(slot), this.balance);
    this.pad = createDpad(this, layout.dpad!, d => this.gameScene.turn(d.x, d.y));
    this.banners = new BannerLayer(this, layout, this.balance);
    this.waitHint = this.add.text(WIDTH / 2, BOARD.y + BOARD.size / 2, this.sys.game?.device?.input?.touch ? TEXT.waitHintTouch : TEXT.waitHintKeys,
      { fontFamily: FONT_UI, fontSize: `${ui(22)}px`, fontStyle: '800', color: TOKENS.text.primary, align: 'center' }).setName('wait-hint').setOrigin(0.5).setVisible(false);
    this.count = new CountdownOverlay(this);
    this.pause = new PauseOverlay(this, { onResume: () => { this.applySettings(); this.gameScene.apply(flowResume(this.gameScene.flow, this.balance)); },
      onSettings: () => this.scene.launch('Settings', { overlay: true }), onQuit: () => { this.session.quit(); this.toResults(0); } }); // MEC-003-R20, MEC-008-R6
    this.createDebugButtons(layout);
    this.applySettings();
    const flow = (e: FlowEffect): void => this.onFlow(e), events = (e: SimEvent[]): void => this.onEvents(e), over = (s: SimSnapshot): void => this.onOver(s);
    const toggle = (): void => this.toggleSound(), pause = (): void => this.gameScene.apply(flowPause(this.gameScene.flow));
    const hidden = (): void => { if (document.hidden) { pause(); this.session.hidden(); } };
    const keys = this.input.keyboard;
    const onKey = (e: KeyboardEvent): void => { const slot = skillSlotForKey(e.code, e.repeat); if (slot === null) return; e.preventDefault(); this.useSkill(slot); }; // MEC-007-R14/R15
    this.gameScene.events.on('flow', flow); this.gameScene.events.on('sim-events', events);
    this.gameScene.events.on('run-over', over); this.gameScene.events.on('toggle-sound', toggle);
    keys?.on('keydown', onKey); window.addEventListener('blur', pause); document.addEventListener('visibilitychange', hidden);
    this.events.once(Phaser.Scenes.Events.SHUTDOWN, () => {
      this.gameScene.events.off('flow', flow); this.gameScene.events.off('sim-events', events);
      this.gameScene.events.off('run-over', over); this.gameScene.events.off('toggle-sound', toggle);
      keys?.off('keydown', onKey); window.removeEventListener('blur', pause); document.removeEventListener('visibilitychange', hidden);
      this.overTimer?.remove(); this.banners.hide();
    });
    this.refresh(); this.showRunStart();
  }

  /** Settings that can change while paused (Settings scene overlay, P3b-3): sound and D-pad. */
  applySettings(): void {
    const s = this.session.save().settings;
    this.synth.setOn(s.sound); this.pad.setVisible(s.dpad);
  }
  private settingsAvailable(): boolean { return this.scene.get('Settings') !== null && this.scene.get('Settings') !== undefined; }

  private drawHill(): void { // GameD2 footer hill (sky gradient), drawn under the controls
    const top = HEIGHT - ui(180), w = ui(530), x = WIDTH / 2 - w / 2;
    this.add.graphics().fillStyle(toInt(TOKENS.sky.top)).fillEllipse(WIDTH / 2, top + ui(90), w, ui(180))
      .fillGradientStyle(toInt(TOKENS.sky.top), toInt(TOKENS.sky.top), toInt(TOKENS.sky.bottom), toInt(TOKENS.sky.bottom), 1)
      .fillRect(x, top + ui(90), w, HEIGHT - top - ui(90));
  }

  // MEC-003 §8 run-start banner. Hud is (re)launched once per run start, after Game emitted newRun, so show it here.
  private showRunStart(): void {
    const start = runStartLines(this.gameScene.snapshot() ?? { level: 0, challenge: false }, t);
    if (start.length) this.banners.show([{ kind: 'challenge', title: start[0]! }]);
  }

  // MEC-003-R21 ?debug=1 only: "+1" and "Key" jump through the normal level-up path; the run becomes a debug run.
  private createDebugButtons(l: ReturnType<typeof hudLayout>): void {
    this.debugKey = this.debugNext = null;
    if (typeof window === 'undefined' || !isDebug(window.location?.search ?? '')) return;
    const skip = (key: boolean): void => {
      if (!acceptSkillInput(this.gameScene.flow.status)) return;
      const cur = this.gameScene.snapshot()?.level; if (cur === undefined) return;
      const to = key ? nextKeyLevel(cur, this.balance) : cur + 1;
      if (to !== null && to <= this.balance.level.maxLevel) this.session.debugSkipTo(to);
    };
    const y = l.chip.y + l.chip.h / 2, h = ui(HUD_MOCK.chip);
    this.debugNext = pillButton(this, ui(16) + ui(28), y, 56, h / ui(1), '+1', 'debug-next', () => skip(false), 'cream');
    this.debugKey = pillButton(this, WIDTH - ui(16) - ui(35), y, 70, h / ui(1), 'Key ▶', 'debug-key', () => skip(true), 'cream');
  }
  private useSkill(slot: 0 | 1): void { if (acceptSkillInput(this.gameScene.flow.status)) this.session.useSkill(slot); } // MEC-007-R15

  private onFlow(effect: FlowEffect): void {
    switch (effect.type) {
      case 'newRun': this.overTimer?.remove(); this.banners.hide(); this.synth.unlock(); this.showRunStart(); break; // MEC-003 §8
      case 'hideOverlay': this.pause.hide(); this.count.hide(); this.waitHint.setVisible(false); break;
      case 'showPaused': this.count.hide(); this.waitHint.setVisible(false); this.showPause(); break;
      case 'tick': {
        const s = this.gameScene.snapshot();
        this.count.tick(effect.n, s ? countdownCard(s, !this.gameScene.flow.started, this.balance, t) : null); this.synth.play('tick'); break;
      }
      case 'go': this.count.hide(); this.waitHint.setVisible(this.gameScene.flow.status === 'waiting'); this.synth.play('go'); break;
      case 'begin': this.waitHint.setVisible(false); break;
    }
    this.refresh();
  }
  private showPause(): void {
    const s = this.gameScene.snapshot(); if (!s) return;
    this.pause.show(pauseView({ level: s.level, biome: s.biome, hint: homeView(this.session.homeModel()).hint, settingsAvailable: this.settingsAvailable() }, t));
  }

  // P1 HUD event mapping (MEC-003/006/007/008 §8) with the P3b-2 banner/tip views.
  private onEvents(events: SimEvent[]): void {
    const ui = this.balance.ui;
    for (const e of events) switch (e.type) {
      case 'ate': this.synth.play(e.golden ? 'golden' : 'eat'); this.synth.buzz(ui.haptics.eatMs); break;
      case 'shrunk': this.shrunkTo = e.from > e.to ? e.to : null; break; // MEC-001-R4 chip only when the snake shrank
      case 'levelUp': this.onLevelUp(e); break;
      case 'runWon': this.synth.play('level'); this.time.delayedCall(ui.bannerMs / 4, () => this.synth.play('level')); break;
      case 'landed': this.synth.play('land'); break;
      case 'rotted': this.synth.play('rot'); break;
      case 'orbSpawned':
        if (!this.orbTipShown && this.bestLevelAtStart < 2) { this.orbTipShown = true; this.banners.tip(t('tip.orb.first')); }
        break;
      case 'orbEaten': this.synth.play('orb'); this.synth.buzz(ui.hud.orbHapticMs); break;
      case 'skillUsed': this.synth.play(SKILL_SFX[e.skill]); this.synth.buzz(ui.hud.castHapticMs); break;
      case 'skillRejected': {
        if (e.reason === 'empty') break;
        this.synth.play('nope'); this.skills.shake(e.slot);
        this.banners.tip(t(`skill.reject.${e.reason}`, { cost: e.skill ? this.balance.skills[e.skill].cost : 0 }));
        break;
      }
      case 'shieldBroke': this.synth.play('shieldBreak'); break;
      case 'skillUnlocked': { const tip = skillTip(e, t); this.banners.tip(tip.text, tip.slot); this.synth.play('unlock'); break; }
      case 'slotUnlocked': this.banners.tip(t('tip.slotUnlocked')); break;
      case 'mangoStolen': this.banners.tip(t('toast.mangoStolen')); break; // MEC-016 §8 toast
      case 'rampEnded': this.synth.play('go'); break;
      case 'died': this.synth.play('die'); this.synth.buzz(ui.haptics.deathMs); break;
      default: break;
    }
    this.refresh();
  }
  private onLevelUp(e: Extract<SimEvent, { type: 'levelUp' }>): void {
    const cps = this.session.save().unlockedCheckpoints;
    let newCheckpoint: number | null = null;
    if (cps.length > this.cpCount) { // MEC-008 §8 checkpoint banner + first-unlock tip
      newCheckpoint = Math.max(...cps);
      if (this.cpCount === 0) this.banners.tip(t('tip.checkpoint'));
      this.cpCount = cps.length;
    }
    this.banners.show(levelUpBanners(e, { shrunkTo: this.shrunkTo, newCheckpoint }, this.balance, t));
    this.shrunkTo = null; this.synth.play('level'); this.synth.buzz(this.balance.ui.haptics.level);
  }

  private onOver(_s: SimSnapshot): void {
    this.banners.hide(); this.count.hide(); this.waitHint.setVisible(false);
    this.toResults(reduceMotion() ? 0 : this.balance.flow.deathCardDelayMs);
  }
  private toResults(delayMs: number): void {
    if (this.leaving) return;
    this.leaving = true; this.overTimer?.remove();
    this.overTimer = this.time.delayedCall(delayMs, () => { this.scene.stop('Game'); this.scene.start('Results'); });
  }
  private toggleSound(): void { // M key (the HUD sound button moved to Settings, P3b-3)
    const on = !this.session.save().settings.sound;
    this.session.setSound(on); this.synth.setOn(on); if (on) this.synth.play('eat');
  }

  update(): void { this.refresh(); }
  private refresh(): void {
    const s = this.gameScene.snapshot(); if (!s) return;
    const status = this.gameScene.flow.status, hint = homeView(this.session.homeModel()).hint; // MEC-008-R11
    const v = hudView(s, hint, this.balance, t);
    this.top.apply(v, status === 'paused', status !== 'ready' && status !== 'over');
    this.skills.apply(v.skills);
    this.banners.setPills(skillPills(s.skills, this.balance, t)); this.banners.setSlowmo(s.slowmo !== null);
    const max = this.balance.level.maxLevel;
    this.debugNext?.c.setVisible(s.level < max); this.debugKey?.c.setVisible(nextKeyLevel(s.level, this.balance) !== null);
    const heading = `${s.heading.x},${s.heading.y}`;
    if (heading !== this.heading) { this.pad.setHeading(s.heading); this.heading = heading; }
  }
}
```
