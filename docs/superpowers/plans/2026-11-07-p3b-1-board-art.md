> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the greybox board with the D2 board art: six biome grounds (subtle checker + grid), light and dark board themes, rocks (pebble/rock/boulder/crumble), warning tiles that are never colour-only, fences, hedges, mud/ice/rising-stone tiles and the mango icon.

## Lead decisions (recorded in DECISIONS in Task 8)
1. **Dark board theme is derived.** The mockups only show light grounds for biomes 1-5 and an already dark Storm Garden. Dark = each light ground mixed 70% toward `#131D15` (the existing dark `bg`), edge colour white, grid white at 10%. Biome 6 keeps its own colours in both themes. Edge-vs-ground contrast must be >= 4.5:1 for every biome and theme (unit tested).
2. **One rock type in core.** Core has a single `rock` hazard; "pebble" and "boulder" exist in the mockup only as sizes. Mapping: hazard/rotten/snip rock = `rock` (0.82), skill rubble = `pebble` (0.5), `boulder` (0.98) is implemented in `drawRock` but unused until a spec uses it.
3. **Mango only.** FoodIcon also shows fish/bone/melon per skin; skins are P4 (spec D8). Only the mango path is ported.
4. **Board outline and screen background are unchanged** (mockup uses a 4 px cream ring on a cream card; the in-game screen background belongs to P3b-2).
5. **Coconut lane band, bomb fuse, cats/chickens** appear on the biome boards but are not in the P3b-1 scope list (spec §4); `movers.ts`, `blasts.ts`, `cycles.ts`, `shrink.ts` are not restyled here (follow-up P3b-1b if the owner wants it).
6. **Approximations** (Graphics has no blur/CSS blob radius): rock blob = ellipse (aspect 0.92); hedge glow = one expanded translucent rounded rect; stone tile asymmetric radii = one radius; "!" and chevron are drawn as shapes (mockup uses Fredoka text/SVG).
7. **Ice and mud warn shapes** are not in the mockups (Tile.dc only has mud, rise-warn, rise-up); the existing drop (mud) and snowflake (ice) shapes are kept and only recoloured.

```ts
import type { BoardTheme } from '../../meta';
import { toInt } from '../uiTokens';

// P3b1-R1 per-biome board look (BiomeBoardsD2). Pure: no Phaser.
export interface BoardLook {
  biome: number; theme: BoardTheme; groundA: number; groundB: number;
  grid: number; gridAlpha: number; edge: number; woodEdge: number;
}
// BiomeBoardsD2 light grounds, one pair per biome (GDD §5.1 order).
const LIGHT_GROUNDS: readonly (readonly [string, string])[] = [
  ['#CFEBD0', '#F4E7A6'], ['#D9EDC2', '#EADFA4'], ['#DCEFFB', '#F5FBFF'],
  ['#EBDDBC', '#DCC79B'], ['#F8D2A6', '#F0BF8A'], ['#3B3363', '#332C58'],
];
const STORM = 6;
const NAVY_EDGE = '#2B3248', WHITE = '#FFFFFF', WOOD_EDGE = '#4A2E1B';
const DARK_BASE = '#131D15', DARK_MIX = 0.7;                 // lead decision 1
const GRID = { light: { color: '#FFFFFF', alpha: 0.5 }, dark: { color: '#FFFFFF', alpha: 0.1 }, storm: { color: '#7CF2D4', alpha: 0.28 } } as const;

export function mixInt(a: number, b: number, t: number): number {
  const ch = (s: number): number => Math.round(((a >> s) & 255) * (1 - t) + ((b >> s) & 255) * t);
  return (ch(16) << 16) | (ch(8) << 8) | ch(0);
}
const channel = (v: number): number => { const c = v / 255; return c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4; };
const luminance = (c: number): number => 0.2126 * channel((c >> 16) & 255) + 0.7152 * channel((c >> 8) & 255) + 0.0722 * channel(c & 255);
export function contrastRatio(a: number, b: number): number {
  const x = luminance(a), y = luminance(b);
  return (Math.max(x, y) + 0.05) / (Math.min(x, y) + 0.05);
}

export function boardLook(biome: number, theme: BoardTheme): BoardLook {
  const n = Math.min(STORM, Math.max(1, Math.round(biome)));
  const [a, b] = LIGHT_GROUNDS[n - 1]!;
  const storm = n === STORM, dark = theme === 'dark' && !storm;
  const groundA = dark ? mixInt(toInt(a), toInt(DARK_BASE), DARK_MIX) : toInt(a);
  const groundB = dark ? mixInt(toInt(b), toInt(DARK_BASE), DARK_MIX) : toInt(b);
  const grid = storm ? GRID.storm : theme === 'dark' ? GRID.dark : GRID.light;
  const edge = toInt(storm || theme === 'dark' ? WHITE : NAVY_EDGE);
  return { biome: n, theme, groundA, groundB, grid: toInt(grid.color), gridAlpha: grid.alpha, edge,
    woodEdge: storm || theme === 'dark' ? edge : toInt(WOOD_EDGE) };
}
export const DEFAULT_LOOK: BoardLook = boardLook(1, 'light');
```

```ts
import { toInt } from '../uiTokens';

// P3b1-R2 board parts from Rock/Tile/FencePiece/FoodIcon/BiomeBoardsD2 (mockup HTML). Pure: no Phaser.
export type RockKind = 'pebble' | 'rock' | 'boulder';
export type Pt = readonly [number, number];

export const PART = {
  rock: { fill: toInt('#9AA3B8'), shade: toInt('#6B7490') },
  wood: { rail: toInt('#E3AC6D'), post: toInt('#CF9150') },
  mud: { fill: toInt('#B8956A'), puddle: toInt('#8F6F43') },
  ice: { fill: toInt('#9ED8F2'), shine: toInt('#B8E5F7') },
  stone: { top: toInt('#EDEFF4'), front: toInt('#A9B0C0'), warnFill: toInt('#D3D7E0'), warnDash: toInt('#FF9A1F'), chevron: toInt('#B35F00') },
  warn: { fill: toInt('#FF6B5B'), dash: toInt('#E5483A'), bang: toInt('#B3261E') },
  hedge: { fill: toInt('#4FBF7A'), edge: toInt('#FFFFFF'), glow: toInt('#7CF2D4') },
  mango: { body: toInt('#FFA21A'), leaf: toInt('#2E9E4F'), hi: toInt('#FFD27A') },
} as const;

/** Fractions of one cell unless noted. */
export const ROCK = { scale: { pebble: 0.5, rock: 0.82, boulder: 0.98 } as Record<RockKind, number>, aspect: 0.92,
  edge: 0.1, band: 0.19, drop: 0.095, dropAlpha: 0.35, hiColor: 0xffffff, hiAlpha: 0.55, hiRotDeg: -20,
  crumbleAlpha: 0.65, previewAlpha: 0.35, crackWidth: 0.075 } as const;
export const WARN = { ring: 0.1, inset: 0.05, radius: 0.3, tileRadius: 0.285, dash: 0.2, dashWidth: 0.095, chevronWidth: 0.13,
  fillBase: 0.3, fillGain: 0.4 } as const;
export const FENCE = { rail: 0.2, railHigh: -0.28, railLow: 0.08, postW: 0.4, postH: 0.7, postRadius: 0.15, edge: 0.076 } as const;
export const TILE = { inset: 1 / 21, radius: 1 / 3, border: 0.095, stoneRadius: 0.2, topShare: 0.68, frontFrom: 0.58 } as const;
export const HEDGE = { bar: 0.8, edge: 0.11, glow: 0.12, glowAlpha: 0.22 } as const;
export const GRID_LINE = 0.1;
export const MANGO = { box: 32, fill: 0.92, body: { cx: 16, cy: 18.5, rx: 11.5, ry: 10, rotDeg: -20 },
  hi: { p: [[8.5, 16], [9.5, 13], [11.7, 11], [14.7, 10.4]] as readonly Pt[], width: 2.4 },
  leaf: [[[16, 9], [17, 4.5], [21.5, 2.5], [26, 3.5]], [[26, 3.5], [25.4, 8.1], [21, 10.7], [16, 9]]] as readonly (readonly Pt[])[] } as const;

/** P3b1-R2 shape id of each warning kind; every warn state is drawn with its shape, not only a colour. */
export const WARN_SHAPES = { hazard: 'bang', rise: 'chevron', mud: 'drop', ice: 'snowflake' } as const;
export type WarnKind = keyof typeof WARN_SHAPES;

/** P3b1-R2 rock body metrics for a cell of `size` px (Rock.dc). dx/dy locate the highlight centre from the rock centre. */
export function rockMetrics(kind: RockKind, size: number) {
  const d = size * ROCK.scale[kind], h = d * ROCK.aspect;
  return { d, h, edge: size * ROCK.edge, band: size * ROCK.band, drop: size * ROCK.drop,
    hl: { dx: -0.15 * d, dy: -0.26 * h, w: 0.3 * d, h: 0.16 * d, rot: (ROCK.hiRotDeg * Math.PI) / 180 } };
}

/** P3b1-R2 "!" drawn as a rounded bar plus a dot, centred in a cell, offsets from the cell top-left. */
export function bangGeometry(cell: number) {
  const w = cell * 0.11, h = cell * 0.34, top = cell * 0.5 - cell * 0.3;
  return { bar: { x: cell / 2 - w / 2, y: top, w, h, r: w / 2 }, dot: { x: cell / 2, y: top + h + cell * 0.1, r: cell * 0.06 } };
}

/** P3b1-R2 rise-warn chevron `M4 12l6-6 6 6` scaled from the 20 px icon box to the cell. */
export function chevronPoints(cell: number): [Pt, Pt, Pt] {
  const s = cell / 20;
  return [[4 * s, 12 * s], [10 * s, 6 * s], [16 * s, 12 * s]];
}

/** P3b1-R2 hedge bars: one inset bar per cell plus a connector towards the right/lower neighbour. */
export function hedgeRects(cells: readonly { x: number; y: number; k: number }[], cell: number): { x: number; y: number; w: number; h: number }[] {
  const has = new Set(cells.map(c => `${c.x},${c.y}`)), out: { x: number; y: number; w: number; h: number }[] = [];
  for (const c of cells) {
    const bar = cell * HEDGE.bar * c.k, cx = (c.x + 0.5) * cell, cy = (c.y + 0.5) * cell;
    out.push({ x: cx - bar / 2, y: cy - bar / 2, w: bar, h: bar });
    if (has.has(`${c.x + 1},${c.y}`)) out.push({ x: cx, y: cy - bar / 2, w: cell, h: bar });
    if (has.has(`${c.x},${c.y + 1}`)) out.push({ x: cx - bar / 2, y: cy, w: bar, h: cell });
  }
  return out;
}

/** P3b1-R2 sample a cubic Bezier into n + 1 points. */
export function cubic(p0: Pt, p1: Pt, p2: Pt, p3: Pt, n: number): Pt[] {
  const out: Pt[] = [];
  for (let i = 0; i <= n; i++) {
    const t = i / n, u = 1 - t, a = u * u * u, b = 3 * u * u * t, c = 3 * u * t * t, d = t * t * t;
    out.push([a * p0[0] + b * p1[0] + c * p2[0] + d * p3[0], a * p0[1] + b * p1[1] + c * p2[1] + d * p3[1]]);
  }
  return out;
}
```

```ts
type PopCell = { x: number; y: number; k: number };

// P3b1-R6 FencePiece.dc: rails run from a post to the right/lower neighbour, posts are drawn after every rail.
function drawFence(g: Graphics, cells: readonly PopCell[], cell: number, look: BoardLook, alpha: number): void {
  const has = new Set(cells.map(c => `${c.x},${c.y}`)), bw = Math.max(1, cell * FENCE.edge), t = cell * FENCE.rail, r = cell * 0.04;
  for (const c of cells) {
    const cx = (c.x + 0.5) * cell, cy = (c.y + 0.5) * cell;
    if (has.has(`${c.x + 1},${c.y}`)) for (const dy of [FENCE.railHigh, FENCE.railLow]) boxed(g, cx, cy + dy * cell, cell, t, r, PART.wood.rail, look.woodEdge, bw, alpha);
    if (has.has(`${c.x},${c.y + 1}`)) for (const dx of [FENCE.railHigh, FENCE.railLow]) boxed(g, cx + dx * cell, cy, t, cell, r, PART.wood.rail, look.woodEdge, bw, alpha);
  }
  for (const c of cells) {
    const w = cell * FENCE.postW * c.k, h = cell * FENCE.postH * c.k;
    boxed(g, (c.x + 0.5) * cell - w / 2, (c.y + 0.5) * cell - h / 2, w, h, cell * FENCE.postRadius, PART.wood.post, look.woodEdge, bw, alpha);
  }
}

// P3b1-R6 BiomeBoardsD2 hedge maze: glowing bars with a white edge (edge pass, then fill pass so joins have no seam).
function drawHedge(g: Graphics, cells: readonly PopCell[], cell: number, look: BoardLook, alpha: number): void {
  const bars = hedgeRects(cells, cell), glow = cell * HEDGE.glow, edge = cell * HEDGE.edge * 0.5, r = cell * 0.22;
  for (const q of bars) g.fillStyle(PART.hedge.glow, HEDGE.glowAlpha * alpha).fillRoundedRect(q.x - glow, q.y - glow, q.w + 2 * glow, q.h + 2 * glow, r + glow);
  for (const q of bars) g.fillStyle(PART.hedge.edge, alpha).fillRoundedRect(q.x, q.y, q.w, q.h, r);
  for (const q of bars) g.fillStyle(PART.hedge.fill, alpha).fillRoundedRect(q.x + edge, q.y + edge, q.w - 2 * edge, q.h - 2 * edge, Math.max(0, r - edge));
}
```

```ts
export function drawTerrains(g: Graphics, terrains: SimSnapshot['terrains'], cell: number, look: BoardLook, b: Balance, reduced: boolean): void {
  for (const terrain of terrains) {
    const a = Math.min(1, terrain.t / b.ui.terrainFadeInMs) * (terrain.t > terrain.lifeMs
      ? Math.max(0, 1 - (terrain.t - terrain.lifeMs) / b.terrain.fadeOutMs) : 1);
    for (const c of terrain.cells) {
      const x = c.x * cell, y = c.y * cell;
      g.fillStyle(PART.stone.front, 0.3 * a).fillRoundedRect(x + cell * 0.1, y + cell * 0.1, cell * 0.8, cell * 0.8, cell * 0.15);
      if (c.vis === 'warn') {
        const downMs = b.terrain.periodMs - b.terrain.upMs - b.terrain.warnMs;
        const ct = Math.max(0, terrain.t - b.terrain.startDelayMs - c.delay) % b.terrain.periodMs - downMs;
        drawWarn(g, c, cell, Math.max(0, ct), Math.min(1, Math.max(0, ct) / b.terrain.warnMs), look, 'rise', true, reduced);

        boxed(g, x + i, y + i + s * TILE.frontFrom, s, s * (1 - TILE.frontFrom), r, PART.stone.front, look.edge, bw, a);
        boxed(g, x + i, y + i, s, s * TILE.topShare, r, PART.stone.top, look.edge, bw, a);
      }
    }
  }
```
