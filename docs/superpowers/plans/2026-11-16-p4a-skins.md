# P4a Skins (MEC-011) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Players can preview, buy (with coins), equip and play with 5 skins; the choice is saved and shown on Home and in the run.

**Architecture:** Buy and equip rules are pure functions in `src/meta/skins.ts` over three new save fields (save v4). The session exposes them; scenes never touch the save. The art layer gets three new skin definitions on the shared rig and a settable active skin; the board look takes the skin's body and outline colours. A new `CharactersScene` (pure layout + view model + thin Phaser scene) is reached from Home. Rare effects are a small pooled particle layer driven by skin data.

**Tech Stack:** TypeScript strict, Phaser 3.90, Vite, Vitest. No new dependency.

**Spec:** `docs/superpowers/specs/2026-11-16-p4a-skins-design.md` (rules `MEC-011-R1..R25`). Mockup: `docs/design/mockups/d3/skins.html` section A (`png/CharactersA.png`).

```ts
// MEC-011 skins: catalogue from the balance, ownership, coins, equip. Pure: no Phaser, no storage.
export type SkinRarity = 'default' | 'common' | 'rare';
export interface SkinsConfig { order: readonly string[]; rare: readonly string[]; prices: { common: number; rare: number } }
export interface SkinState { coins: number; ownedSkins: string[]; equippedSkin: string }
export type BuyResult = { ok: true; state: SkinState } | { ok: false; reason: 'owned' | 'coins' | 'unknown' };

export const DEFAULT_SKIN = 'snake';
// MEC-011-R5 each call owns its array.
export const defaultSkinState = (): SkinState => ({ coins: 0, ownedSkins: [DEFAULT_SKIN], equippedSkin: DEFAULT_SKIN });

// MEC-011-R2 the default skin is free; every other skin is common unless the balance lists it as rare.
export function skinRarity(cfg: SkinsConfig, id: string): SkinRarity {
  return id === DEFAULT_SKIN ? 'default' : cfg.rare.includes(id) ? 'rare' : 'common';
}
export function skinPrice(cfg: SkinsConfig, id: string): number {
  const r = skinRarity(cfg, id);
  return r === 'default' ? 0 : cfg.prices[r];
}

// MEC-011-R6 buy: known, not owned, enough coins; pays, owns and equips in one step.
export function buySkin(state: SkinState, id: string, cfg: SkinsConfig): BuyResult {
  if (!cfg.order.includes(id)) return { ok: false, reason: 'unknown' };
  if (state.ownedSkins.includes(id)) return { ok: false, reason: 'owned' };
  const price = skinPrice(cfg, id);
  if (state.coins < price) return { ok: false, reason: 'coins' };
  return { ok: true, state: { coins: state.coins - price, ownedSkins: [...state.ownedSkins, id], equippedSkin: id } };
}

// MEC-011-R7 equip: owned skins only; equipping the equipped skin returns the same state.
export function equipSkin(state: SkinState, id: string): { ok: boolean; state: SkinState } {
  if (!state.ownedSkins.includes(id)) return { ok: false, state };
  return { ok: true, state: state.equippedSkin === id ? state : { ...state, equippedSkin: id } };
}

// MEC-011-R10 whole coins, never negative.
export function addCoins(state: SkinState, n: number): SkinState {
  return { ...state, coins: Math.max(0, state.coins + Math.floor(n)) };
}

// MEC-011-R12 load-time repair; owned skins keep the balance order with the default first.
export function sanitizeSkins(raw: { coins?: unknown; ownedSkins?: unknown; equippedSkin?: unknown }, cfg: SkinsConfig): SkinState {
  const coins = typeof raw.coins === 'number' && Number.isFinite(raw.coins) ? Math.max(0, Math.floor(raw.coins)) : 0;
  const listed = Array.isArray(raw.ownedSkins) ? raw.ownedSkins : [];
  const owned = cfg.order.filter(id => id === DEFAULT_SKIN || listed.includes(id));
  const ownedSkins = [DEFAULT_SKIN, ...owned.filter(id => id !== DEFAULT_SKIN)];
  const equippedSkin = typeof raw.equippedSkin === 'string' && ownedSkins.includes(raw.equippedSkin) ? raw.equippedSkin : DEFAULT_SKIN;
  return { coins, ownedSkins, equippedSkin };
}
```

import type { SkinRarity } from '../../meta';
import type { SkinsModel } from '../../session';
import { t } from '../text';

export type TokenState = 'equipped' | 'owned' | 'locked';
export interface CharactersView {
  coins: number; preview: string; name: string; rarity: SkinRarity; perk: string;
  tokens: { id: string; state: TokenState; price: number; selected: boolean }[];
  action: { kind: 'equipped' | 'equip' | 'buy' | 'need'; price: number; missing: number };
}
// MEC-011-R14..R17 everything the Characters screen shows, from the session model and the previewed id.
export function charactersView(m: SkinsModel, preview: string): CharactersView {
  const cur = m.skins.find(s => s.id === preview) ?? m.skins.find(s => s.id === m.equipped)!;
  const kind = cur.id === m.equipped ? 'equipped' : cur.owned ? 'equip' : m.coins >= cur.price ? 'buy' : 'need';
  return {
    coins: m.coins, preview: cur.id, name: t(`skin.${cur.id}.name`), rarity: cur.rarity, perk: t(`skin.${cur.id}.perk`),
    tokens: m.skins.map(s => ({ id: s.id, price: s.price, selected: s.id === cur.id, state: s.id === m.equipped ? 'equipped' : s.owned ? 'owned' : 'locked' })),
    action: { kind, price: cur.price, missing: kind === 'need' ? cur.price - m.coins : 0 },
  };
}

```ts
import type { Balance } from '../../core';

type Cfg = Balance['juice']['skinFx'];
export interface Particle { x: number; y: number; vx: number; vy: number; born: number; life: number; size: number; kind: 'spark' | 'leaf'; rot: number }
export interface SkinFxState { cfg: Cfg; eat: boolean; trail: boolean; reduced: boolean; live: Particle[]; lastLeaf: number }

// MEC-011-R22..R25 cosmetic particles of a Rare skin, in board cells; positions are cell units, times are ms.
export function createSkinFx(cfg: Cfg, fx: { eat: boolean; trail: boolean }, reduced: boolean): SkinFxState {
  return { cfg, eat: fx.eat, trail: fx.trail, reduced, live: [], lastLeaf: Number.NEGATIVE_INFINITY };
}
// MEC-011-R22 gold sparks fly out from the mouth.
export function skinFxEat(s: SkinFxState, now: number, at: { x: number; y: number }, rnd: () => number): void {
  if (!s.eat || s.reduced) return;
  const c = s.cfg.eat;
  for (let i = 0; i < c.count; i++) {
    const a = (i + rnd()) / c.count * Math.PI * 2, v = c.spreadCells / c.lifeMs;
    s.live.push({ x: at.x, y: at.y, vx: Math.cos(a) * v, vy: Math.sin(a) * v, born: now, life: c.lifeMs, size: c.sizeCells * (0.7 + 0.6 * rnd()), kind: 'spark', rot: a });
  }
}
// MEC-011-R23 one leaf per everyMs behind the tail while moving; the oldest leaf is dropped at the cap.
export function skinFxTrail(s: SkinFxState, now: number, tail: { x: number; y: number }, moving: boolean, rnd: () => number): void {
  if (!s.trail || s.reduced || !moving || now - s.lastLeaf < s.cfg.trail.everyMs) return;
  const c = s.cfg.trail; s.lastLeaf = now;
  const leaves = s.live.filter(p => p.kind === 'leaf');
  if (leaves.length >= c.max) s.live.splice(s.live.indexOf(leaves[0]!), 1);
  s.live.push({ x: tail.x, y: tail.y, vx: (rnd() - 0.5) * c.driftCells / c.lifeMs, vy: c.driftCells / c.lifeMs, born: now, life: c.lifeMs, size: c.sizeCells, kind: 'leaf', rot: rnd() * Math.PI });
}
// Drops dead particles; the returned array is the live list (do not keep it across frames).
export function skinFxStep(s: SkinFxState, now: number): readonly Particle[] {
  s.live = s.live.filter(p => now - p.born <= p.life);
  return s.live;
}
```
